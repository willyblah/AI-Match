# 统一收单交易退款接口代码示例

## 请求示例

```typescript
import { AlipaySdk } from "../_shared/alipay-sdk-deno.ts";

const sdk = new AlipaySdk({
  appId: Deno.env.get("ALIPAY_WEB_APP_ID")!,
  privateKey: Deno.env.get("ALIPAY_WEB_PRIVATE_KEY")!,
  alipayPublicKey: Deno.env.get("ALIPAY_WEB_PUBLIC_KEY")!,
  signType: "RSA2",});

const result = await sdk.exec("alipay.trade.refund", {
  bizContent: {
    out_trade_no: "20150320010101001",
    trade_no: "2014112611001004680073956707",
    refund_amount: "200.12",
    refund_reason: "正常退款",
    out_request_no: "HZ01RF001",
    refund_goods_detail: [
      {
        out_sku_id: "outSku_01",
        out_item_id: "outItem_01",
        goods_id: "apple-01",
        refund_amount: "19.50",
        out_certificate_no_list: ["202407013232143241231243243423"],
      },
    ],
    query_options: ["refund_detail_item_list"],
  },}, isProduction);

if (result.code === "10000") {
  console.log("调用成功", result);
} else {
  console.log("调用失败", result.subMsg);
}
```
